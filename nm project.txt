<script>

  /*
   * ============================================================
   * GEMINI CONFIGURATION
   * ============================================================
   *
   * For a quick local demo, replace the value below with your
   * Gemini API key.
   *
   * IMPORTANT:
   * Never expose an API key like this in a production application.
   * Use a backend endpoint instead.
   */

  const GEMINI_API_KEY = "YOUR_GEMINI_API_KEY";

  /*
   * Gemini REST endpoint.
   *
   * Replace the model name with a currently available Gemini model
   * in your Google AI environment if necessary.
   */

  const GEMINI_MODEL = "gemini-2.5-flash";

  const GEMINI_URL =
    `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${GEMINI_API_KEY}`;


  async function generatePlan() {

    hideError();

    const profile = getProfile();

    if (!validateProfile(profile)) {
      return;
    }

    if (
      GEMINI_API_KEY === "YOUR_GEMINI_API_KEY" ||
      !GEMINI_API_KEY.trim()
    ) {
      showError(
        "Please add your Gemini API key to GEMINI_API_KEY in the JavaScript section."
      );
      return;
    }

    document.getElementById("loading").style.display = "block";
    document.getElementById("result").style.display = "none";

    const prompt = createPrompt(profile);

    try {

      const response = await fetch(GEMINI_URL, {

        method: "POST",

        headers: {
          "Content-Type": "application/json"
        },

        body: JSON.stringify({

          contents: [
            {
              role: "user",
              parts: [
                {
                  text: prompt
                }
              ]
            }
          ],

          generationConfig: {
            temperature: 0.7,
            maxOutputTokens: 5000
          }

        })

      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          data?.error?.message ||
          "Gemini API request failed."
        );
      }

      const text =
        data?.candidates?.[0]?.content?.parts
          ?.map(part => part.text || "")
          .join("") || "";

      if (!text) {
        throw new Error("Gemini returned an empty response.");
      }

      const cleaned = cleanJson(text);

      let plan;

      try {
        plan = JSON.parse(cleaned);
      } catch (jsonError) {
        throw new Error(
          "The AI returned an unexpected format. Please try generating the plan again."
        );
      }

      renderPlan(plan, profile);

      document.getElementById("result").style.display = "block";

      saveProfile(profile);
      savePlan(plan);

    } catch (error) {

      console.error(error);

      showError(
        error.message ||
        "Something went wrong while generating your fitness plan."
      );

    } finally {

      document.getElementById("loading").style.display = "none";
    }
  }


  function getProfile() {

    return {

      name:
        document.getElementById("name").value.trim(),

      age:
        document.getElementById("age").value.trim(),

      gender:
        document.getElementById("gender").value,

      goal:
        document.getElementById("goal").value,

      level:
        document.getElementById("level").value,

      days:
        document.getElementById("days").value,

      duration:
        document.getElementById("duration").value,

      equipment:
        document.getElementById("equipment").value.trim(),

      restrictions:
        document.getElementById("restrictions").value.trim(),

      preferences:
        document.getElementById("preferences").value.trim()

    };

  }


  function validateProfile(profile) {

    if (!profile.name) {
      showError("Please enter your name.");
      return false;
    }

    if (!profile.age) {
      showError("Please enter your age.");
      return false;
    }

    if (profile.age < 13 || profile.age > 100) {
      showError("Please enter an age between 13 and 100.");
      return false;
    }

    return true;
  }


  function createPrompt(profile) {

    return `
You are FitBuddy, an AI fitness planning assistant.

Create a personalized ${profile.days}-day fitness plan for this user.

USER PROFILE
------------
Name: ${profile.name}
Age: ${profile.age}
Gender: ${profile.gender || "Not specified"}
Goal: ${profile.goal}
Fitness level: ${profile.level}
Workout days per week: ${profile.days}
Workout duration: ${profile.duration}
Equipment: ${profile.equipment || "No equipment"}
Restrictions/injuries: ${profile.restrictions || "None stated"}
Exercise preferences: ${profile.preferences || "None stated"}

IMPORTANT SAFETY RULES
----------------------
- Do not diagnose medical conditions.
- Do not claim that the plan is medically appropriate.
- If the user reports an injury, medical issue, or restriction, avoid exercises that could aggravate it and recommend professional guidance where appropriate.
- Do not prescribe medication or supplements.
- Keep exercises appropriate for the stated fitness level.
- Include warm-up and cool-down.
- Provide alternatives for exercises when practical.
- Nutrition information should be general educational guidance, not medical treatment.
- Do not make extreme calorie restriction recommendations.

OUTPUT
------
Return ONLY valid JSON.

Use exactly this structure:

{
  "title": "string",
  "summary": "string",
  "weekly_goal": "string",
  "days": [
    {
      "day": "string",
      "focus": "string",
      "warmup": "string",
      "exercises": [
        {
          "name": "string",
          "sets": "string",
          "reps": "string",
          "rest": "string",
          "instructions": "string",
          "alternative": "string"
        }
      ],
      "cooldown": "string"
    }
  ],
  "nutrition": {
    "general_guidance": "string",
    "breakfast": "string",
    "lunch": "string",
    "dinner": "string",
    "snacks": "string",
    "hydration": "string"
  },
  "tips": [
    "string",
    "string",
    "string",
    "string"
  ]
}

Create practical, understandable and realistic recommendations.
`;
  }


  function cleanJson(text) {

    text = text.trim();

    if (text.startsWith("```")) {

      text = text
        .replace(/^```json/i, "")
        .replace(/^```/i, "")
        .replace(/```$/i, "")
        .trim();

    }

    const firstBrace = text.indexOf("{");
    const lastBrace = text.lastIndexOf("}");

    if (firstBrace !== -1 && lastBrace !== -1) {
      text = text.substring(firstBrace, lastBrace + 1);
    }

    return text;
  }


  function renderPlan(plan, profile) {

    const container =
      document.getElementById("planContent");

    let html = "";

    html += `
      <div class="day">
        <h3>${escapeHtml(plan.title || "Personalized Fitness Plan")}</h3>

        <p class="exercise-details">
          ${escapeHtml(plan.summary || "")}
        </p>

        ${
          plan.weekly_goal
            ? `
              <p class="exercise-details" style="margin-top:10px;">
                <strong>Weekly Goal:</strong>
                ${escapeHtml(plan.weekly_goal)}
              </p>
            `
            : ""
        }
      </div>
    `;


    if (Array.isArray(plan.days)) {

      plan.days.forEach(day => {

        html += `
          <div class="day">

            <h3>
              ${escapeHtml(day.day || "Workout")}
            </h3>

            <p class="exercise-details">
              <strong>Focus:</strong>
              ${escapeHtml(day.focus || "")}
            </p>

            <p class="exercise-details"
               style="margin:12px 0;">
              <strong>🔥 Warm-up:</strong>
              ${escapeHtml(day.warmup || "")}
            </p>
        `;


        if (Array.isArray(day.exercises)) {

          day.exercises.forEach(exercise => {

            html += `
              <div class="exercise">

                <div class="exercise-name">
                  ${escapeHtml(exercise.name || "Exercise")}
                </div>

                <div class="exercise-details">

                  <strong>Sets:</strong>
                  ${escapeHtml(exercise.sets || "-")}

                  &nbsp; | &nbsp;

                  <strong>Reps:</strong>
                  ${escapeHtml(exercise.reps || "-")}

                  &nbsp; | &nbsp;

                  <strong>Rest:</strong>
                  ${escapeHtml(exercise.rest || "-")}

                  <br>

                  <strong>How:</strong>
                  ${escapeHtml(exercise.instructions || "")}

                  ${
                    exercise.alternative
                      ? `
                        <br>
                        <strong>Alternative:</strong>
                        ${escapeHtml(exercise.alternative)}
                      `
                      : ""
                  }

                </div>

              </div>
            `;

          });

        }


        html += `
            <p class="exercise-details"
               style="margin-top:12px;">
              <strong>🧘 Cool-down:</strong>
              ${escapeHtml(day.cooldown || "")}
            </p>

          </div>
        `;

      });

    }


    if (plan.nutrition) {

      const nutrition = plan.nutrition;

      html += `
        <div class="nutrition">

          <h3>🥗 Nutrition Guidance</h3>

          <p class="meal">
            <strong>General:</strong>
            ${escapeHtml(nutrition.general_guidance || "")}
          </p>

          <p class="meal">
            <strong>Breakfast:</strong>
            ${escapeHtml(nutrition.breakfast || "")}
          </p>

          <p class="meal">
            <strong>Lunch:</strong>
            ${escapeHtml(nutrition.lunch || "")}
          </p>

          <p class="meal">
            <strong>Dinner:</strong>
            ${escapeHtml(nutrition.dinner || "")}
          </p>

          <p class="meal">
            <strong>Snacks:</strong>
            ${escapeHtml(nutrition.snacks || "")}
          </p>

          <p class="meal">
            <strong>Hydration:</strong>
            ${escapeHtml(nutrition.hydration || "")}
          </p>

        </div>
      `;

    }


    if (Array.isArray(plan.tips)) {

      html += `
        <div class="tips">

          <h3>💡 FitBuddy Tips</h3>

          <ul>
      `;

      plan.tips.forEach(tip => {

        html += `
          <li>${escapeHtml(tip)}</li>
        `;

      });

      html += `
          </ul>

        </div>
      `;

    }


    container.innerHTML = html;

    window.scrollTo({
      top: document.getElementById("result").offsetTop - 20,
      behavior: "smooth"
    });

  }


  function escapeHtml(value) {

    if (value === null || value === undefined) {
      return "";
    }

    return String(value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");

  }


  function showError(message) {

    const error =
      document.getElementById("error");

    error.textContent = message;
    error.style.display = "block";

    error.scrollIntoView({
      behavior: "smooth",
      block: "center"
    });

  }


  function hideError() {

    document.getElementById("error").style.display =
      "none";

  }


  function clearForm() {

    document.getElementById("name").value = "";
    document.getElementById("age").value = "";
    document.getElementById("gender").value = "";
    document.getElementById("goal").value = "Weight loss";
    document.getElementById("level").value = "Beginner";
    document.getElementById("days").value = "3";
    document.getElementById("duration").value = "45 minutes";
    document.getElementById("equipment").value = "";
    document.getElementById("restrictions").value = "";
    document.getElementById("preferences").value = "";

    document.getElementById("result").style.display = "none";

    hideError();

  }


  function saveProfile(profile) {

    try {
      localStorage.setItem(
        "fitbuddy_profile",
        JSON.stringify(profile)
      );
    } catch (e) {
      console.warn("Could not save profile.");
    }

  }


  function savePlan(plan) {

    try {
      localStorage.setItem(
        "fitbuddy_plan",
        JSON.stringify(plan)
      );
    } catch (e) {
      console.warn("Could not save plan.");
    }

  }


  function loadSavedData() {

    try {

      const savedProfile =
        localStorage.getItem("fitbuddy_profile");

      if (savedProfile) {

        const profile =
          JSON.parse(savedProfile);

        document.getElementById("name").value =
          profile.name || "";

        document.getElementById("age").value =
          profile.age || "";

        document.getElementById("gender").value =
          profile.gender || "";

        document.getElementById("goal").value =
          profile.goal || "Weight loss";

        document.getElementById("level").value =
          profile.level || "Beginner";

        document.getElementById("days").value =
          profile.days || "3";

        document.getElementById("duration").value =
          profile.duration || "45 minutes";

        document.getElementById("equipment").value =
          profile.equipment || "";

        document.getElementById("restrictions").value =
          profile.restrictions || "";

        document.getElementById("preferences").value =
          profile.preferences || "";

      }


      const savedPlan =
        localStorage.getItem("fitbuddy_plan");

      if (savedPlan) {

        const plan =
          JSON.parse(savedPlan);

        renderPlan(
          plan,
          JSON.parse(savedProfile || "{}")
        );

        document.getElementById("result").style.display =
          "block";
      }

    } catch (e) {

      console.warn(
        "Saved FitBuddy data could not be loaded."
      );

    }

  }


  window.addEventListener(
    "DOMContentLoaded",
    loadSavedData
  );
